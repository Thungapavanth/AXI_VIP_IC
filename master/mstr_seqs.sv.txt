class mstr_sequence extends uvm_sequence#(axi_xtn);
	`uvm_object_utils(mstr_sequence)
	

	function new(string name="mstr_sequence");
		super.new(name);
	endfunction

endclass


class fixed extends mstr_sequence;
	`uvm_object_utils(fixed)

	axi_xtn xtn;
	function new(string name= "fixed");
		super.new(name);
	endfunction


	task body();
		repeat(4)
			begin
				xtn=axi_xtn::type_id::create("xtn");
				start_item(xtn);
				assert(xtn.randomize() with {AWBURST==2'b00;AWADDR=='d4;AWSIZE==2; ARLEN==3;});
				finish_item(xtn);
			end
	endtask
endclass
